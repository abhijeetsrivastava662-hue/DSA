1class Solution {
2public:
3    int countValidPrefixes(string s) {
4        int zero=0;
5        int one =0;
6        int count=0;
7        for(int i=0;i<s.size();i++){
8            if(s[i]=='0') zero++;
9            else if(s[i]=='1') one++;
10            if(abs(zero-one)<=1) count++;
11        }
12        return count;
13    }
14};
